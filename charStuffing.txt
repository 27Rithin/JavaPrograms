import java.io.*;
 
  public class CharStuffing
{
  public static void main(String[] args) throws IOException
{

  BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

  System.out.println("Enter the message:");
  String input = br.readLine();

  String stuffedData = "DLESTX";

  for(int i = 0; i < input.length(); i++)
{
   
   if(i + 2 < input.length()

      &&(input.charAt(i) == 'D' || input.charAt(i) == 'd')
      &&(input.charAt(i+1) == 'L' || input.charAt(i+1) == 'l')
      &&(input.charAt(i+2) == 'E' || input.charAt(i+2) == 'e')

       stuffedData += "DLE";
}

       stuffedData += input.charAt(i);
  
       stuffedData += "DLEETX";

       System.out.println("stuffed Output: " + stuffedData);
  
}
}
