alter:

Modifying Tables (Most Common Use):

-- Add a new column
ALTER TABLE employees ADD COLUMN email VARCHAR(100);

-- Drop a column
ALTER TABLE employees DROP COLUMN phone_number;

-- Modify a column's data type
ALTER TABLE employees ALTER COLUMN salary DECIMAL(10,2);

-- Rename a column
ALTER TABLE employees RENAME COLUMN emp_name TO full_name;

-- Add a constraint
ALTER TABLE employees ADD CONSTRAINT pk_emp_id PRIMARY KEY (emp_id);

-- Drop a constraint
ALTER TABLE employees DROP CONSTRAINT pk_emp_id;


Modifying Other Database Objects
-- Rename a table
ALTER TABLE old_table_name RENAME TO new_table_name;

-- Modify a view
ALTER VIEW employee_view AS SELECT * FROM employees WHERE active = 1;

-- Alter indexes
ALTER INDEX idx_name REBUILD;


1. Adding New Columns

-- Add a column for student's nationality
ALTER TABLE students ADD COLUMN nationality VARCHAR(50) AFTER gender;

-- Add a column with default value
ALTER TABLE students ADD COLUMN is_active BOOLEAN DEFAULT TRUE AFTER enrollment_date;


2. Modifying Existing Columns

-- Change column data type
ALTER TABLE students MODIFY COLUMN phone_number VARCHAR(20);

-- Rename a column
ALTER TABLE students CHANGE COLUMN First_name first_name VARCHAR(50);

-- Add NOT NULL constraint
ALTER TABLE students MODIFY COLUMN email VARCHAR(100) NOT NULL;

-- Change ENUM values
ALTER TABLE students MODIFY COLUMN gender ENUM('MALE','FEMALE','OTHER','PREFER_NOT_TO_SAY');






4. Table Structure Changes

-- Rename table
ALTER TABLE students RENAME TO university_students;

-- Change storage engine
ALTER TABLE students ENGINE = InnoDB;

-- Modify character set
ALTER TABLE students CONVERT TO CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;




3. Constraints Management
-- Add primary key (though you already have one)
ALTER TABLE students ADD PRIMARY KEY (student_id);

-- Add foreign key (if relating to another table)
ALTER TABLE students ADD COLUMN department_id INT;
ALTER TABLE students ADD CONSTRAINT fk_department 
    FOREIGN KEY (department_id) REFERENCES departments(department_id);

-- Add unique constraint
ALTER TABLE students ADD CONSTRAINT uc_phone UNIQUE (phone_number);



5. Dropping Columns/Constraints

-- Drop a column
ALTER TABLE students DROP COLUMN is_active;

-- Drop a constraint
ALTER TABLE students DROP FOREIGN KEY fk_department;
ALTER TABLE students DROP INDEX uc_phone;



6. Partitioning (Advanced)

-- Add partitioning by year of enrollment
ALTER TABLE students PARTITION BY RANGE (YEAR(enrollment_date)) (
    PARTITION p2020 VALUES LESS THAN (2021),
    PARTITION p2021 VALUES LESS THAN (2022),
    PARTITION p2022 VALUES LESS THAN (2023),
    PARTITION pmax VALUES LESS THAN MAXVALUE
);


7. Modifying Default Values
-- Change default value
ALTER TABLE students ALTER COLUMN gender SET DEFAULT 'PREFER_NOT_TO_SAY';

-- Remove default value
ALTER TABLE students ALTER COLUMN gender DROP DEFAULT;


8. Index Operations
-- Add index
ALTER TABLE students ADD INDEX idx_last_name (Last_name);

-- Add fulltext index for searching
ALTER TABLE students ADD FULLTEXT INDEX ft_search (first_name, Last_name, address);

-- Drop index
ALTER TABLE students DROP INDEX idx_last_name;

Remember to back up your database before performing major ALTER operations, especially those that modify column types or drop columns. Some operations like changing data types or adding constraints may fail if existing data violates the new rules.
